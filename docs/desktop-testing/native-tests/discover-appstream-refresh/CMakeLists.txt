cmake_minimum_required(VERSION 3.20)
project(RealPackageKitRefreshTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Gui DBus Test Concurrent Qml)
find_package(AppStreamQt REQUIRED)
find_package(packagekitqt6 REQUIRED)
find_package(KF6Service REQUIRED)
set(SOURCE /root/build/plasma-discover-6.7.4)
set(BUILD ${SOURCE}/obj-aarch64-linux-gnu)
foreach(variant baseline fixed)
 if(variant STREQUAL baseline)
  set(HEADERS /root/build/fix3-source/discover-6.7.4)
  set(LIBS /root/build/discover-fix3-verify/usr/lib/aarch64-linux-gnu)
  set(COMMON ${LIBS}/plasma-discover/libDiscoverCommon.so)
  set(PLUGIN ${LIBS}/qt6/plugins/discover/packagekit-backend.so)
 else()
  if(DEFINED CANDIDATE_HEADERS)
   set(HEADERS ${CANDIDATE_HEADERS})
  else()
   set(HEADERS ${SOURCE})
  endif()
  if(DEFINED CANDIDATE_LIBRARY_ROOT)
   set(COMMON ${CANDIDATE_LIBRARY_ROOT}/plasma-discover/libDiscoverCommon.so)
   set(PLUGIN ${CANDIDATE_LIBRARY_ROOT}/qt6/plugins/discover/packagekit-backend.so)
  else()
   set(COMMON ${BUILD}/bin/libDiscoverCommon.so)
   set(PLUGIN ${BUILD}/bin/discover/packagekit-backend.so)
  endif()
 endif()
 add_executable(backend-test-${variant} main.cpp)
 target_include_directories(backend-test-${variant} PRIVATE ${HEADERS}/libdiscover/backends/PackageKitBackend ${HEADERS}/libdiscover ${BUILD}/libdiscover ${BUILD})
 target_link_libraries(backend-test-${variant} PRIVATE Qt6::Core Qt6::Gui Qt6::DBus Qt6::Test Qt6::Concurrent Qt6::Qml KF6::Service PK::packagekitqt6 AppStreamQt ${COMMON} "-Wl,${PLUGIN}")
 if(variant STREQUAL fixed)
  target_compile_definitions(backend-test-${variant} PRIVATE CANDIDATE=1)
 endif()
endforeach()
