cmake_minimum_required(VERSION 3.16)
project(termux-usb-bridge VERSION 0.4.1 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
find_package(Threads REQUIRED)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(termux-usb-bridge SHARED src/client.c)
target_include_directories(termux-usb-bridge PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_options(termux-usb-bridge PRIVATE -Wall -Wextra -Werror)
set_target_properties(termux-usb-bridge PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
add_library(termux-usb-gamepad SHARED src/adapters/hid.c src/adapters/ipc.c src/adapters/gamepad_client.c)
target_include_directories(termux-usb-gamepad PUBLIC include PRIVATE src/adapters)
target_link_libraries(termux-usb-gamepad PRIVATE Threads::Threads)
add_executable(tusb-probe src/probe.c)
target_link_libraries(tusb-probe PRIVATE termux-usb-bridge)
option(TUSB_WITH_LIBUSB "Build a real libusb wrap_sys_device probe (Linux/Android only)" OFF)
option(TUSB_ADAPTERS "Build all UVC/PipeWire, SDL2/3 and filesystem adapters" OFF)
option(TUSB_FILESYSTEM "Build USB BOT/FAT/exFAT filesystem adapter" OFF)
option(TUSB_DESKTOP "Build GIO removable volumes and desktop hotplug coordinator" OFF)
option(TUSB_CAMERA2 "Build Android Camera2 capture and PipeWire publication" OFF)
option(TUSB_GSTREAMER "Build private upstream GStreamer PipeWire plugin" OFF)
if(TUSB_CAMERA2 OR TUSB_GSTREAMER)
    find_package(PkgConfig REQUIRED)
endif()
if(TUSB_CAMERA2)
    if(NOT ANDROID)
        message(FATAL_ERROR "TUSB_CAMERA2 requires Android/Bionic; containers consume the PipeWire source")
    endif()
    pkg_check_modules(CAMERA2_PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=0.3.49)
    add_executable(tusb-camera2-capture src/adapters/camera2.c)
    target_link_libraries(tusb-camera2-capture PRIVATE camera2ndk mediandk ${CMAKE_DL_LIBS} Threads::Threads)
    add_executable(tusb-camera2-source src/adapters/camera2_source.c)
    target_link_libraries(tusb-camera2-source PRIVATE PkgConfig::CAMERA2_PIPEWIRE Threads::Threads)
    foreach(target tusb-camera2-capture tusb-camera2-source)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endforeach()
    install(TARGETS tusb-camera2-capture tusb-camera2-source RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/termux-usb-bridge)
    install(FILES camera2.py DESTINATION ${CMAKE_INSTALL_DATADIR}/termux-usb-bridge)
endif()
if(TUSB_GSTREAMER)
    pkg_check_modules(GST_PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3 gstreamer-video-1.0
        gstreamer-audio-1.0 gstreamer-allocators-1.0 gstreamer-net-1.0)
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/generated/pipewire/config.h
        "#define PACKAGE \"pipewire\"\n#define PACKAGE_VERSION \"1.4.9\"\n#define HAVE_GSTREAMER_DEVICE_PROVIDER 1\n")
    file(GLOB GST_PIPEWIRE_SOURCES CONFIGURE_DEPENDS third_party/pipewire-gst/gst/*.c)
    add_library(tusb-gstpipewire MODULE ${GST_PIPEWIRE_SOURCES})
    set_target_properties(tusb-gstpipewire PROPERTIES OUTPUT_NAME gstpipewire)
    target_include_directories(tusb-gstpipewire PRIVATE third_party/pipewire-gst ${CMAKE_CURRENT_BINARY_DIR}/generated/pipewire)
    target_link_libraries(tusb-gstpipewire PRIVATE PkgConfig::GST_PIPEWIRE m)
    install(TARGETS tusb-gstpipewire LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/termux-usb-bridge/gstreamer-1.0)
    install(FILES third_party/pipewire-gst/COPYING DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/termux-usb-bridge RENAME COPYING.pipewire)
endif()
if(TUSB_WITH_LIBUSB OR TUSB_ADAPTERS OR TUSB_FILESYSTEM)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0>=1.0.24)
    if(TUSB_WITH_LIBUSB)
        target_link_libraries(tusb-probe PRIVATE PkgConfig::LIBUSB)
        target_compile_definitions(tusb-probe PRIVATE TUSB_WITH_LIBUSB)
    endif()
endif()
if(TUSB_DESKTOP)
    if(NOT TUSB_ADAPTERS)
        message(FATAL_ERROR "TUSB_DESKTOP requires TUSB_ADAPTERS")
    endif()
    pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0 gio-unix-2.0 gmodule-2.0)
    add_library(tusb-volume-monitor MODULE src/desktop/volume_monitor.c)
    target_link_libraries(tusb-volume-monitor PRIVATE PkgConfig::GIO)
    target_compile_options(tusb-volume-monitor PRIVATE -Wall -Wextra -Werror)
    add_executable(tusb-gio-mount src/adapters/gio_mount.c)
    target_link_libraries(tusb-gio-mount PRIVATE PkgConfig::GIO)
    add_executable(tusb-inspect src/adapters/inspect.c)
    target_link_libraries(tusb-inspect PRIVATE termux-usb-bridge PkgConfig::LIBUSB)
    install(TARGETS tusb-volume-monitor LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/gio/modules)
    install(TARGETS tusb-gio-mount tusb-inspect RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES desktop.py DESTINATION ${CMAKE_INSTALL_DATADIR}/termux-usb-bridge)
    if(BUILD_TESTING)
        add_executable(tusb-desktop-probe tests/desktop_probe.c)
        target_link_libraries(tusb-desktop-probe PRIVATE PkgConfig::GIO)
    endif()
endif()
if(TUSB_FILESYSTEM OR TUSB_ADAPTERS)
    add_library(termux-usb-fs SHARED src/adapters/filesystem.c third_party/fatfs/ff.c third_party/fatfs/ffunicode.c)
    target_include_directories(termux-usb-fs PUBLIC include PRIVATE third_party/fatfs)
    target_link_libraries(termux-usb-fs PRIVATE termux-usb-bridge PkgConfig::LIBUSB Threads::Threads)
    install(TARGETS termux-usb-fs LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(FILES include/termux_usb_fs.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
endif()
if(TUSB_ADAPTERS)
    pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=0.3.49)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.24)
    pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3>=3.2)
    find_package(JPEG REQUIRED)
    pkg_check_modules(TURBOJPEG REQUIRED IMPORTED_TARGET libturbojpeg)
    set(libuvc_VERSION_MAJOR 0)
    set(libuvc_VERSION_MINOR 0)
    set(libuvc_VERSION_PATCH 7)
    set(libuvc_VERSION 0.0.7)
    set(LIBUVC_HAS_JPEG TRUE)
    configure_file(third_party/libuvc/include/libuvc/libuvc_config.h.in generated/libuvc/libuvc_config.h @ONLY)
    add_library(tusb-uvc STATIC
        third_party/libuvc/src/ctrl.c third_party/libuvc/src/ctrl-gen.c
        third_party/libuvc/src/device.c third_party/libuvc/src/diag.c
        third_party/libuvc/src/frame.c third_party/libuvc/src/frame-mjpeg.c
        third_party/libuvc/src/init.c third_party/libuvc/src/stream.c third_party/libuvc/src/misc.c)
    target_include_directories(tusb-uvc PUBLIC third_party/libuvc/include ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(tusb-uvc PUBLIC PkgConfig::LIBUSB JPEG::JPEG Threads::Threads)
    add_executable(tusb-camera src/adapters/camera.c src/adapters/camera_frame.c)
    target_link_libraries(tusb-camera PRIVATE termux-usb-bridge tusb-uvc PkgConfig::PIPEWIRE PkgConfig::TURBOJPEG)
    add_executable(tusb-gamepadd src/adapters/gamepadd.c src/adapters/ipc.c)
    target_include_directories(tusb-gamepadd PRIVATE include src/adapters)
    target_link_libraries(tusb-gamepadd PRIVATE termux-usb-gamepad termux-usb-bridge PkgConfig::LIBUSB Threads::Threads)
    foreach(version 2 3)
        add_library(termux-usb-sdl${version} SHARED src/adapters/sdl_adapter.c)
        target_include_directories(termux-usb-sdl${version} PRIVATE include)
        target_link_libraries(termux-usb-sdl${version} PRIVATE termux-usb-gamepad PkgConfig::SDL${version} ${CMAKE_DL_LIBS} Threads::Threads)
        if(version EQUAL 3)
            target_compile_definitions(termux-usb-sdl${version} PRIVATE TUSB_SDL3)
        endif()
        install(TARGETS termux-usb-sdl${version} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
        if(BUILD_TESTING)
            add_executable(tusb-sdl${version}-probe tests/sdl_probe.c)
            target_link_libraries(tusb-sdl${version}-probe PRIVATE PkgConfig::SDL${version})
            if(version EQUAL 3)
                target_compile_definitions(tusb-sdl${version}-probe PRIVATE TUSB_SDL3)
            endif()
        endif()
    endforeach()
    install(TARGETS tusb-camera tusb-gamepadd RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES third_party/libuvc/LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/termux-usb-bridge RENAME COPYING.libuvc)
    if(BUILD_TESTING)
        add_executable(tusb-camera-codec tests/camera_codec.c src/adapters/camera_frame.c)
        target_include_directories(tusb-camera-codec PRIVATE src/adapters)
        target_link_libraries(tusb-camera-codec PRIVATE PkgConfig::TURBOJPEG)
        add_executable(tusb-camera-sink tests/camera_sink.c)
        target_link_libraries(tusb-camera-sink PRIVATE PkgConfig::PIPEWIRE)
    endif()
endif()
if(ANDROID)
    set(TUSB_SHELL "${CMAKE_INSTALL_PREFIX}/bin/sh")
else()
    set(TUSB_SHELL /bin/sh)
endif()
configure_file(scripts/termux-usb-bridge.in termux-usb-bridge @ONLY)
configure_file(scripts/termux-usb-run.in termux-usb-run @ONLY)
configure_file(scripts/tusb-files.in tusb-files @ONLY)
configure_file(scripts/tusb-sdl-run.in tusb-sdl-run @ONLY)
configure_file(scripts/tusb-desktop.in tusb-desktop @ONLY)
configure_file(scripts/tusb-camera2.in tusb-camera2 @ONLY)
configure_file(scripts/tusb-camera-run.in tusb-camera-run @ONLY)
configure_file(desktop/tusb-desktop.desktop.in tusb-desktop.desktop @ONLY)
configure_file(termux-usb-bridge.pc.in termux-usb-bridge.pc @ONLY)
install(TARGETS termux-usb-bridge termux-usb-gamepad tusb-probe
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/termux_usb_bridge.h include/termux_usb_gamepad.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES broker.py filesystem.py DESTINATION ${CMAKE_INSTALL_DATADIR}/termux-usb-bridge)
install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/termux-usb-bridge
    ${CMAKE_CURRENT_BINARY_DIR}/termux-usb-run DESTINATION ${CMAKE_INSTALL_BINDIR})
if(TUSB_ADAPTERS)
    install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/tusb-sdl-run DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(TUSB_CAMERA2)
    install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/tusb-camera2 DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(TUSB_GSTREAMER)
    install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/tusb-camera-run DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(TUSB_DESKTOP)
    install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/tusb-desktop DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/tusb-desktop.desktop DESTINATION ${CMAKE_INSTALL_SYSCONFDIR}/xdg/autostart)
endif()
if(TUSB_FILESYSTEM OR TUSB_ADAPTERS)
    install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/tusb-files DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES third_party/fatfs/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/termux-usb-bridge RENAME COPYING.fatfs)
endif()
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/termux-usb-bridge.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME broker-integration COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_bridge.py $<TARGET_FILE:termux-usb-bridge> $<TARGET_FILE:tusb-probe>)
    add_test(NAME hid-parser COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_hid.py $<TARGET_FILE:termux-usb-gamepad>)
    if(TUSB_FILESYSTEM OR TUSB_ADAPTERS)
        add_test(NAME filesystem-webdav COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_filesystem.py $<TARGET_FILE:termux-usb-fs>)
    endif()
    if(TUSB_ADAPTERS)
        add_test(NAME sdl-gamepad COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sdl.py ${CMAKE_CURRENT_BINARY_DIR})
        add_test(NAME pipewire-camera COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_camera.py ${CMAKE_CURRENT_BINARY_DIR})
        add_test(NAME camera-codecs COMMAND tusb-camera-codec)
    endif()
endif()
